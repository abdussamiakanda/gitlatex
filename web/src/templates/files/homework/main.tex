\begin{problem}[1]
Prove that $\sqrt{2}$ is irrational.
\end{problem}
\begin{solution}
Suppose $\sqrt{2} = p/q$ in lowest terms. Then $p^2 = 2q^2$, so $p$ is even \ldots
\end{solution}

  \item $\displaystyle \int_0^1 x^2\,dx$
\end{enumerate}
\end{problem}
